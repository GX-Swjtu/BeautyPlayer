function(copy_gp_runtime target)
    # Both executables share a runtime directory. One producer prevents parallel
    # post-link copies from colliding on Windows file locks.
    if(NOT TARGET gpupixel_runtime)
        add_custom_target(gpupixel_runtime
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GP_ROOT}/library/windows/gpupixel.dll" "${CMAKE_BINARY_DIR}"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GP_ROOT}/library/windows/mars-face-kit.dll" "${CMAKE_BINARY_DIR}"
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${GP_ROOT}/models" "${CMAKE_BINARY_DIR}/models"
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${GP_ROOT}/res" "${CMAKE_BINARY_DIR}/res")
    endif()
    add_dependencies(${target} gpupixel_runtime)
endfunction()
